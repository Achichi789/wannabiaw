-- ============================================================================ -- SUPABASE DATABASE SCHEMA FOR MเบียวTI (แบบทดสอบวัดระดับความเบียวออนไลน์) -- รองรับการเก็บข้อมูลครบทั้ง 4 ส่วนตามโครงสร้างแอปพลิเคชัน -- ============================================================================ -- เปิดใช้งาน Extension สำหรับสร้าง UUID อัตโนมัติ CREATE EXTENSION IF NOT EXISTS "uuid-ossp"; -- ============================================================================ -- 1. ตารางข้อมูล User ก่อนทำการวัดระดับ (users_profile) -- เก็บข้อมูลผู้ใช้งานเบื้องต้น นามแฝง/ฉายา และข้อมูลอุปกรณ์ก่อนเริ่มทำแบบประเมิน -- ============================================================================ CREATE TABLE public.users_profile ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), codename VARCHAR(100) NOT NULL, -- นามแฝง / ฉายาประจำตัวที่กรอกหน้าแรก device_type VARCHAR(50), -- ประเภทอุปกรณ์ เช่น mobile, desktop user_agent TEXT, -- ข้อมูลเบราว์เซอร์ created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); COMMENT ON TABLE public.users_profile IS 'ส่วนที่ 1: เก็บข้อมูล User ก่อนทำการวัดระดับความเบียว'; -- ============================================================================ -- 2. ตารางระดับความเบียวหลังการวัดระดับ (assessment_results) -- เก็บผลลัพธ์คะแนนรวม รหัสบุคลิกภาพ MเบียวTI และฉายาที่ได้รับหลังทำแบบทดสอบเสร็จ -- ============================================================================ CREATE TABLE public.assessment_results ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID NOT NULL REFERENCES public.users_profile(id) ON DELETE CASCADE, total_score INTEGER NOT NULL CHECK (total_score >= 8 AND total_score <= 56), max_score INTEGER NOT NULL DEFAULT 56, mbiawti_code VARCHAR(20) NOT NULL, -- เช่น 'NORM-1', 'COOL-4', 'CHUN-7', 'GOD-99' persona_title VARCHAR(150) NOT NULL, -- เช่น 'มหาเทพผู้บัญชาการโชคชะตา' aura_level VARCHAR(100), -- เช่น '9,999% (ไร้เทียมทาน)' completed_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); COMMENT ON TABLE public.assessment_results IS 'ส่วนที่ 2: เก็บผลลัพธ์ระดับความเบียวและรหัสบุคลิกภาพหลังการวัดระดับ'; -- ============================================================================ -- 3. ตารางข้อมูลความชอบ ความสนใจ เรื่องต่างๆ ในแบบสอบถาม (user_preferences_answers) -- เก็บคำตอบรายข้อ (สถานการณ์ชีวิตประจำวัน, บทสนทนา, กิจกรรม, อาหาร, ซีรีส์, สไตล์ไอเทม) -- ============================================================================ CREATE TABLE public.user_preferences_answers ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), assessment_id UUID NOT NULL REFERENCES public.assessment_results(id) ON DELETE CASCADE, user_id UUID NOT NULL REFERENCES public.users_profile(id) ON DELETE CASCADE, question_id INTEGER NOT NULL, -- ลำดับข้อคำถาม (1 - 8) category VARCHAR(100) NOT NULL, -- หมวดหมู่ เช่น 'อาหารและเครื่องดื่มที่ชอบทาน', 'ซีรีส์และภาพยนตร์ที่สนใจ' question_type VARCHAR(20) NOT NULL CHECK (question_type IN ('choice', 'slider', 'checkbox')), selected_options JSONB NOT NULL, -- เก็บรายละเอียดตัวเลือกที่เลือก เช่น ["fashion", "gadget"] หรือระดับ 1-7 question_score INTEGER NOT NULL, -- คะแนนที่ได้ในข้อนั้น (1 - 7) created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); COMMENT ON TABLE public.user_preferences_answers IS 'ส่วนที่ 3: เก็บข้อมูลความชอบ ความสนใจ และคำตอบรายข้อในแบบสอบถาม'; -- ============================================================================ -- 4. ตารางข้อมูล Transaction ความสนใจในผลิตภัณฑ์ที่แนะนำ (product_recommendation_transactions) -- เก็บข้อมูลว่าหลังวัดระดับแล้ว User สนใจสินค้าที่เว็บแนะนำหรือไม่ และคลิกลิงก์สินค้าชิ้นใดบ้าง -- ============================================================================ -- ตารางอ้างอิงรายการสินค้าในระบบ (Master Products) CREATE TABLE public.recommended_products_catalog ( product_id VARCHAR(50) PRIMARY KEY, -- เช่น 'PROD-FASH-01', 'PROD-GADG-02' category_tag VARCHAR(50) NOT NULL, -- 'fashion', 'gadget', 'beverage', 'lifestyle' product_name VARCHAR(200) NOT NULL, price NUMERIC(10, 2) NOT NULL, product_url TEXT NOT NULL, is_active BOOLEAN DEFAULT TRUE ); -- เพิ่มข้อมูลสินค้าตั้งต้นให้ตรงกับในหน้าเว็บ INSERT INTO public.recommended_products_catalog (product_id, category_tag, product_name, price, product_url) VALUES ('PROD-FASH-01', 'fashion', 'เสื้อคลุมทรง Oversized สีดำรัตติกาล & แหวนเงินสลักลาย', 890.00, 'https://example.com/shop/dark-overcoat-ring?ref=mbiawti'), ('PROD-GADG-02', 'gadget', 'หูฟังไร้สายตัดเสียงรบกวน (Soundtrack Mode Edition)', 1590.00, 'https://example.com/shop/anc-headphones?ref=mbiawti'), ('PROD-BEV-03', 'beverage', 'ชุดดริปกาแฟ Dark Abyss Roast & แก้วทัมเบลอร์สีดำด้าน', 550.00, 'https://example.com/shop/dark-abyss-coffee?ref=mbiawti'), ('PROD-LIFE-04', 'lifestyle', 'สมุดบันทึกปกหนังแผนการลับ & ปากกาหมึกเจลสีดำสนิท', 320.00, 'https://example.com/shop/strategy-notebook?ref=mbiawti'); -- ตารางเก็บ Transaction การตอบสนองต่อสินค้าแนะนำของผู้ใช้แต่ละคน CREATE TABLE public.product_recommendation_transactions ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), assessment_id UUID NOT NULL REFERENCES public.assessment_results(id) ON DELETE CASCADE, user_id UUID NOT NULL REFERENCES public.users_profile(id) ON DELETE CASCADE, is_interested BOOLEAN, -- TRUE = สนใจ, FALSE = ไม่สนใจ, NULL = ยังไม่ได้ระบุ shown_product_ids JSONB NOT NULL, -- รายการรหัสสินค้าทั้งหมดที่ระบบแนะนำให้ user คนนี้เห็น clicked_products JSONB DEFAULT '[]'::jsonb, -- รายการสินค้าที่ user กดคลิกลิงก์สั่งซื้อ พร้อมเวลาที่คลิก updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); COMMENT ON TABLE public.product_recommendation_transactions IS 'ส่วนที่ 4: เก็บข้อมูล Transaction ว่าหลังวัดระดับ User สนใจสินค้าที่แนะนำหรือไม่ และสนใจชิ้นใดบ้าง'; -- ============================================================================ -- 5. ตาราง LEADERBOARD (leaderboard) -- ตารางสรุปอันดับคะแนนรวม รหัสบุคลิกภาพ และสถานะความสนใจสินค้าสำหรับแสดงผลหน้าเว็บแบบรวดเร็ว -- ============================================================================ CREATE TABLE public.leaderboard ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), user_id UUID REFERENCES public.users_profile(id) ON DELETE CASCADE, assessment_id UUID REFERENCES public.assessment_results(id) ON DELETE CASCADE, codename VARCHAR(100) NOT NULL, total_score INTEGER NOT NULL, mbiawti_code VARCHAR(20) NOT NULL, persona_title VARCHAR(150) NOT NULL, is_interested_product BOOLEAN, interested_products JSONB DEFAULT '[]'::jsonb, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); COMMENT ON TABLE public.leaderboard IS 'ตารางจัดอันดับคะแนนความเบียวรวม (Leaderboard) เชื่อมโยงกับผลลัพธ์ทั้งหมด'; -- ============================================================================ -- INDEXES & ROW LEVEL SECURITY (RLS) สำหรับ SUPABASE -- ============================================================================ CREATE INDEX idx_results_user_id ON public.assessment_results(user_id); CREATE INDEX idx_answers_assessment_id ON public.user_preferences_answers(assessment_id); CREATE INDEX idx_transactions_assessment_id ON public.product_recommendation_transactions(assessment_id); CREATE INDEX idx_leaderboard_score ON public.leaderboard(total_score DESC, created_at ASC); -- เปิดใช้งาน RLS เพื่อความปลอดภัยบน Supabase ALTER TABLE public.users_profile ENABLE ROW LEVEL SECURITY; ALTER TABLE public.assessment_results ENABLE ROW LEVEL SECURITY; ALTER TABLE public.user_preferences_answers ENABLE ROW LEVEL SECURITY; ALTER TABLE public.recommended_products_catalog ENABLE ROW LEVEL SECURITY; ALTER TABLE public.product_recommendation_transactions ENABLE ROW LEVEL SECURITY; ALTER TABLE public.leaderboard ENABLE ROW LEVEL SECURITY; -- สร้าง Policy อนุญาตให้ผู้เล่นแบบ Anonymous (Public Web) สามารถบันทึกและอ่านข้อมูลได้ CREATE POLICY "Allow public insert users_profile" ON public.users_profile FOR INSERT WITH CHECK (true); CREATE POLICY "Allow public insert assessment_results" ON public.assessment_results FOR INSERT WITH CHECK (true); CREATE POLICY "Allow public insert user_preferences_answers" ON public.user_preferences_answers FOR INSERT WITH CHECK (true); CREATE POLICY "Allow public select products_catalog" ON public.recommended_products_catalog FOR SELECT USING (true); CREATE POLICY "Allow public insert transactions" ON public.product_recommendation_transactions FOR INSERT WITH CHECK (true); CREATE POLICY "Allow public update transactions" ON public.product_recommendation_transactions FOR UPDATE USING (true); CREATE POLICY "Allow public select leaderboard" ON public.leaderboard FOR SELECT USING (true); CREATE POLICY "Allow public insert leaderboard" ON public.leaderboard FOR INSERT WITH CHECK (true); CREATE POLICY "Allow public update leaderboard" ON public.leaderboard FOR UPDATE USING (true);
